  Raw vs. derived data

Raw data

General recommendations1

  • Save and backup the raw data
  • Do not modify raw data (even for minor changes)
  • Raw data should be in a read-only mode (🔒)
  • Any modification produces an output or a derived data
  • Write code for data acquisition (when possible)
    • Database queries
    • API requests
    • download.file(), wget, curl, etc.
  • Describe and document raw data (README, metadata, etc.)

  1. Wilson et al. (2017) https://doi.org/10.1371/journal.pcbi.1005510

Sorting and arranging


R package dplyr is very useful to:


  • select columns (based on their names only, no $var / [, "var"] / [["var"]]): dplyr::select(column1, column2)

  • filter rows from those selected columns: dplyr::filter(column1 == a, column2 > x)

  • group and aggregate rows: group_by(factor_column)

  • create new columns: mutate(new_column = old_column * 2)

  • always work with data.frame or tibble

Part 1 - Clean herbarium data

## Import herbarium data ----
herbarium <- read.csv2("UC1-3c-open-refine.csv")

## Clean data ----
herbarium <- herbarium |>

  janitor::clean_names() |>

  dplyr::select(country_col, full_name, lat, lon) |>

  dplyr::mutate(country_col = stringr::str_squish(country_col)) |>
  dplyr::mutate(full_name = stringr::str_squish(full_name)) |>

  dplyr::mutate(lon = parzer::parse_lon(lon)) |>
  dplyr::mutate(lat = parzer::parse_lat(lat)) |>

  dplyr::filter(
    full_name %in% c("Catopsis sessiliflora", "Catopsis nutans")
  )


## Check data ----
anyNA(herbarium$country_col)
unique(herbarium$country_col)

anyNA(herbarium$full_name)
unique(herbarium$full_name)

anyNA(herbarium$lon)
range(herbarium$lon)

anyNA(herbarium$lat)
range(herbarium$lat)

Part 2 - Clean GBIF occurrences

## Import GBIF occurrences data ----
gbif_occ <- readxl::read_xlsx("gbif_occurrences.xlsx")

## Clean data ----
gbif_occ <- gbif_occ |>

  janitor::clean_names() |>

  dplyr::mutate(
    scientific_name = stringr::str_replace(
      scientific_name,
      "^([[:alpha:]]+\\s+[[:alpha:]]+).*",
      "\\1"
    )
  )


## Check data ----
anyNA(gbif_occ$scientific_name)
unique(gbif_occ$scientific_name)

anyNA(gbif_occ$country)
unique(gbif_occ$country)

anyNA(gbif_occ$decimal_longitude)
range(gbif_occ$decimal_longitude)

anyNA(gbif_occ$decimal_latitude)
range(gbif_occ$decimal_latitude)

Part 3 - Row bind the two datasets

## Rename herbarium columns ----
herbarium <- herbarium |>
  dplyr::rename(
    country = country_col,
    scientific_name = full_name,
    decimal_latitude = lat,
    decimal_longitude = lon
  )


## Row binds the two datasets ----
final <- dplyr::bind_rows(herbarium, gbif_occ)

Correction (bonus)

Part 4 - Check coordinates

## Filter coordinates based on Guatemala extent ----
final <- final |>

  dplyr::filter(
    decimal_longitude >= -92.22925,
    decimal_longitude <= -88.22502,
    decimal_latitude >= 13.73534,
    decimal_latitude <= 17.81933
  )


Part 5 - Map occurrences

## Convert data.frame into sf object ----
final_sf <- final |>
  sf::st_as_sf(
    coords = c("decimal_longitude", "decimal_latitude"),
    crs = 4326
  )

## Get Guatemala boundaries ----
gtm <- rnaturalearth::ne_countries(country = "Guatemala")

## Plot map ----
ggplot2::ggplot(data = gtm) +
  ggplot2::geom_sf() +

  ggplot2::geom_sf(
    data = final_sf,
    mapping = ggplot2::aes(color = scientific_name)
  ) +

  ggplot2::theme_minimal()
